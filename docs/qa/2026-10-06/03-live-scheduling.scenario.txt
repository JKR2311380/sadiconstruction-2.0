const p=q.live.p;
q.addNodeUI=async(p,kind,name,duration,parent)=>{await p.getByRole('button',{name:kind==='summary'?'+ Nested summary':'+ Activity',exact:true}).click();const d=p.getByRole('dialog');if(parent){await d.getByRole('combobox').click();await p.getByRole('option',{name:new RegExp(parent)}).click();}await d.getByLabel('Name',{exact:true}).fill(name);if(kind==='leaf')await d.getByLabel('Duration (working days)',{exact:true}).fill(String(duration));await d.getByRole('button',{name:'Add',exact:true}).click();await d.waitFor({state:'hidden'});await p.waitForTimeout(800);};
q.row=(p,name)=>p.locator('section[aria-label="Activity schedule and Gantt chart"] tbody tr').filter({has:p.getByRole('button',{name:new RegExp(name)})});
q.openPredUI=async(p,name)=>{await q.row(p,name).locator('td').nth(2).getByRole('button').click();await p.getByRole('dialog',{name:'Edit predecessors'}).waitFor();};
q.addLinkUI=async(p,target,pred,type,lag)=>{await q.openPredUI(p,target);const d=p.getByRole('dialog');await d.getByRole('combobox').nth(0).click();await p.getByRole('option',{name:new RegExp(pred)}).click();await d.getByRole('combobox').nth(1).click();await p.getByRole('option',{name:type,exact:true}).click();await d.getByLabel('Lag (working days)',{exact:true}).fill(String(lag));await d.getByRole('button',{name:'Add link',exact:true}).click();await d.getByRole('button',{name:'Done',exact:true}).click();};
await q.test('LIVE-SCHEDULE-ADD-SUMMARY',async()=>{await q.addNodeUI(p,'summary','QA Package');q.assert(await q.row(p,'QA Package').count()===1,'Summary missing');return 'Nested summary created under Phase A';});
await q.test('LIVE-SCHEDULE-ADD-LEAF',async()=>{await q.addNodeUI(p,'leaf','QA Foundation',5,'QA Package');await q.addNodeUI(p,'leaf','QA Followup',3,'QA Package');q.assert(await q.row(p,'QA Foundation').count()===1,'Leaf missing');return {evidence:await q.snap(p,'qa-activities-created')};});
await q.test('LIVE-SCHEDULE-FS-LAG',async()=>{await q.addLinkUI(p,'QA Followup','QA Foundation','FS',2);q.assert((await p.locator('footer').innerText()).includes('10 working days'),'5 + FS lag2 +3 should be 10');return {rows:await p.locator('section[aria-label="Activity schedule and Gantt chart"] tbody').innerText(),evidence:await q.snap(p,'qa-fs-lag')};});
await q.test('LIVE-SCHEDULE-BASELINE',async()=>{await p.getByRole('button',{name:'Set baseline',exact:true}).click();await p.getByRole('button',{name:'Re-baseline',exact:true}).waitFor();q.assert((await p.locator('footer').innerText()).includes('on baseline'),'Baseline comparison absent');return 'Baseline captured at 10 days';});
await q.test('LIVE-SCHEDULE-DURATION-RECALC',async()=>{const field=p.getByRole('spinbutton',{name:/QA Foundation duration in working days/});await field.fill('8');await field.press('Enter');q.assert((await p.locator('footer').innerText()).includes('13 working days'),'Duration should become 13');q.assert((await p.locator('footer').innerText()).includes('+3 working days'),'Baseline delta should be +3');return {evidence:await q.snap(p,'qa-duration-baseline')};});
await q.test('LIVE-SCHEDULE-PROGRESS',async()=>{const field=p.getByRole('spinbutton',{name:/QA Foundation progress/});await field.fill('50');await field.press('Enter');return {footer:await p.locator('footer').innerText(),fields:await p.getByRole('spinbutton').evaluateAll(els=>els.map(e=>({name:e.getAttribute('aria-label'),value:e.value}))),evidence:await q.snap(p,'qa-progress')};});
await p.waitForTimeout(1500);await q.test('LIVE-SCHEDULE-PERSIST-RELOAD',async()=>{await p.reload();await p.getByRole('button',{name:'Re-baseline',exact:true}).waitFor({timeout:15000});q.assert((await p.locator('footer').innerText()).includes('13 working days'),'Duration/link not persisted');q.assert(await p.getByRole('spinbutton',{name:/QA Foundation duration in working days/}).inputValue()==='8','Duration lost');return {evidence:await q.snap(p,'qa-reloaded-schedule')};});
await q.test('LIVE-SCHEDULE-CRITICAL-FILTER',async()=>{await p.getByRole('button',{name:'Critical only',exact:true}).click();q.assert(await p.getByRole('button',{name:'Critical only',exact:true}).getAttribute('aria-pressed')==='true','Critical filter state not updated');await p.getByRole('button',{name:'Critical only',exact:true}).click();return 'Critical-only toggles';});
await q.test('LIVE-SCHEDULE-COLLAPSE',async()=>{await p.getByRole('button',{name:/Toggle .*QA Package/}).click();q.assert(await q.row(p,'QA Foundation').count()===0,'Collapsed child still rendered');await p.getByRole('button',{name:/Toggle .*QA Package/}).click();q.assert(await q.row(p,'QA Foundation').count()===1,'Expanded child missing');return 'Collapse/expand hides/restores descendants';});
await q.test('LIVE-SCHEDULE-ZOOM',async()=>{const chart=p.locator('section[aria-label="Activity schedule and Gantt chart"] > div');const before=await chart.evaluate(e=>e.getBoundingClientRect().width);await p.getByRole('button',{name:'Zoom timeline in',exact:true}).click();const after=await chart.evaluate(e=>e.getBoundingClientRect().width);q.assert(after>before,'Zoom did not widen chart');return {before,after};});
return q.results.slice(-10);
